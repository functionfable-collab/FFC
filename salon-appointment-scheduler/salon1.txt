#!/bin/bash

PSQL="psql --username=freecodecamp --dbname=salon -t -c"

# Function to display services
show_services() {
  echo "Here are our services:"
  $PSQL "SELECT service_id, name FROM services ORDER BY service_id;" | awk -F '|' '{printf "%s) %s\n", $1, $2}'
}

# Show services and get valid service selection
while true
do
  show_services
  read -p "Which service would you like? " SERVICE_ID_SELECTED
  SERVICE_NAME=$($PSQL "SELECT name FROM services WHERE service_id = $SERVICE_ID_SELECTED;" | xargs)
  if [[ ! -z $SERVICE_NAME ]]
  then
    break
  fi
done

# Get phone number
read -p "What's your phone number? " CUSTOMER_PHONE

# Check if customer exists
CUSTOMER_ID=$($PSQL "SELECT customer_id FROM customers WHERE phone = '$CUSTOMER_PHONE';" | xargs)

if [[ -z $CUSTOMER_ID ]]
then
  # New customer - get name and insert
  read -p "What's your name? " CUSTOMER_NAME
  $PSQL "INSERT INTO customers(name, phone) VALUES('$CUSTOMER_NAME', '$CUSTOMER_PHONE');"
  CUSTOMER_ID=$($PSQL "SELECT customer_id FROM customers WHERE phone = '$CUSTOMER_PHONE';" | xargs)
else
  # Existing customer - retrieve their name
  CUSTOMER_NAME=$($PSQL "SELECT name FROM customers WHERE phone = '$CUSTOMER_PHONE';" | xargs)
fi

# Get appointment time
read -p "What time would you like your appointment? " SERVICE_TIME

# Insert appointment
$PSQL "INSERT INTO appointments(customer_id, service_id, time) VALUES($CUSTOMER_ID, $SERVICE_ID_SELECTED, '$SERVICE_TIME');"

# Confirmation message
echo "I have put you down for a $SERVICE_NAME at $SERVICE_TIME, $CUSTOMER_NAME"
